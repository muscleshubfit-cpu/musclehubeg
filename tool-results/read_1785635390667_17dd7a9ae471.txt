   975→const dictionaries: Record<Lang, Dict> = { ar, en };
   976→
   977→type I18nContextValue = {
   978→  lang: Lang;
   979→  dir: "rtl" | "ltr";
   980→  setLang: (l: Lang) => void;
   981→  toggle: () => void;
   982→  t: (key: string) => string;
   983→};
   984→
   985→const I18nContext = createContext<I18nContextValue | null>(null);
   986→
   987→const STORAGE_KEY = "nutra-lang";
   988→
   989→export function I18nProvider({ children }: { children: ReactNode }) {
   990→  const [lang, setLangState] = useState<Lang>("ar");
   991→
   992→  useEffect(() => {
   993→    const stored = (typeof window !== "undefined" &&
   994→      localStorage.getItem(STORAGE_KEY)) as Lang | null;
   995→    if (stored === "ar" || stored === "en") setLangState(stored);
   996→  }, []);
   997→
   998→  const dir: "rtl" | "ltr" = lang === "ar" ? "rtl" : "ltr";
   999→
  1000→  useEffect(() => {
  1001→    if (typeof document === "undefined") return;
  1002→    document.documentElement.lang = lang;
  1003→    document.documentElement.dir = dir;
  1004→  }, [lang, dir]);
  1005→
  1006→  const setLang = (l: Lang) => {
  1007→    setLangState(l);
  1008→    if (typeof window !== "undefined") localStorage.setItem(STORAGE_KEY, l);
  1009→  };
  1010→
  1011→  const toggle = () => setLang(lang === "ar" ? "en" : "ar");
  1012→
  1013→  const t = (key: string) => dictionaries[lang][key] ?? dictionaries.en[key] ?? key;
  1014→
  1015→  return (
  1016→    <I18nContext.Provider value={{ lang, dir, setLang, toggle, t }}>
  1017→      {children}
  1018→    </I18nContext.Provider>
  1019→  );
  1020→}
  1021→
  1022→export function useI18n() {
  1023→  const ctx = useContext(I18nContext);
  1024→  if (!ctx) throw new Error("useI18n must be used within I18nProvider");
  1025→  return ctx;
  1026→}
